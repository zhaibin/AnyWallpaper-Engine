cmake_minimum_required(VERSION 3.14)
set(PROJECT_NAME "anywp_engine")
project(${PROJECT_NAME} LANGUAGES CXX)

# This value is used when generating builds using this plugin, so it must
# not be changed
set(PLUGIN_NAME "anywp_engine_plugin")

# ==========================================
# Precompiled DLL Mode
# ==========================================
# This CMakeLists.txt is for precompiled package distribution
# It uses IMPORTED library instead of compiling from source

# Paths to precompiled binaries (relative to this CMakeLists.txt)
set(PRECOMPILED_DLL "${CMAKE_CURRENT_SOURCE_DIR}/../bin/anywp_engine_plugin.dll")
set(PRECOMPILED_LIB "${CMAKE_CURRENT_SOURCE_DIR}/../lib/anywp_engine_plugin.lib")

if(NOT EXISTS "${PRECOMPILED_DLL}")
  message(FATAL_ERROR "Precompiled DLL not found at ${PRECOMPILED_DLL}")
endif()

if(NOT EXISTS "${PRECOMPILED_LIB}")
  message(FATAL_ERROR "Precompiled LIB not found at ${PRECOMPILED_LIB}")
endif()

# Create imported library (GLOBAL makes it visible in all scopes)
add_library(${PLUGIN_NAME} SHARED IMPORTED GLOBAL)
set_target_properties(${PLUGIN_NAME} PROPERTIES
  IMPORTED_LOCATION "${PRECOMPILED_DLL}"
  IMPORTED_IMPLIB "${PRECOMPILED_LIB}"
)

# Include directories (for C API header)
target_include_directories(${PLUGIN_NAME} INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/../include"
)

# Windows SDK libraries
# Note: flutter and flutter_wrapper_plugin are NOT linked for precompiled packages
# The DLL is already compiled with all required dependencies
target_link_libraries(${PLUGIN_NAME} INTERFACE
  shlwapi
  version
)

# List of absolute paths to libraries that should be bundled with the plugin
set(anywp_engine_bundled_libraries
  "${PRECOMPILED_DLL}"
  "${CMAKE_CURRENT_SOURCE_DIR}/../bin/WebView2Loader.dll"
  PARENT_SCOPE
)

# ==========================================
# Copy SDK file to runtime directory
# This ensures the SDK is available when C++ plugin tries to load it
# ==========================================
# Path to SDK file in precompiled package (v2.2.0+: unified sdk/dist/ structure)
set(ANYWP_SDK_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../sdk/dist/anywp_sdk.js")

# Function to copy SDK file (can be called from main CMakeLists.txt)
# Usage: anywp_copy_sdk(TARGET_NAME)
function(anywp_copy_sdk TARGET_NAME)
  if(EXISTS "${ANYWP_SDK_SOURCE}")
    # Copy SDK to runner directory (where application runs)
    # This is the most important path - where the executable actually runs
    add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory 
        "$<TARGET_FILE_DIR:${TARGET_NAME}>/sdk/dist"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ANYWP_SDK_SOURCE}"
        "$<TARGET_FILE_DIR:${TARGET_NAME}>/sdk/dist/anywp_sdk.js"
      COMMENT "Copying AnyWP SDK to runner directory ($<CONFIG>)"
      VERBATIM
    )
    
    # Also copy to root sdk/ directory for backward compatibility
    # This ensures old integration code still works
    add_custom_command(TARGET ${TARGET_NAME} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E make_directory 
        "$<TARGET_FILE_DIR:${TARGET_NAME}>/sdk"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${ANYWP_SDK_SOURCE}"
        "$<TARGET_FILE_DIR:${TARGET_NAME}>/sdk/anywp_sdk.js"
      COMMENT "Copying AnyWP SDK to sdk/ directory for compatibility ($<CONFIG>)"
      VERBATIM
    )
    
    message(STATUS "AnyWP SDK copy commands added for target: ${TARGET_NAME}")
  else()
    message(WARNING "AnyWP SDK file not found at ${ANYWP_SDK_SOURCE} - SDK injection may fail at runtime")
  endif()
endfunction()

# Check if SDK file exists and provide info
if(EXISTS "${ANYWP_SDK_SOURCE}")
  message(STATUS "AnyWP SDK found at: ${ANYWP_SDK_SOURCE}")
  message(STATUS "Note: Call anywp_copy_sdk(YOUR_TARGET_NAME) in your main CMakeLists.txt to copy SDK to runtime")
else()
  message(WARNING "AnyWP SDK file not found at ${ANYWP_SDK_SOURCE}")
endif()

message(STATUS "Using precompiled AnyWP Engine DLL: ${PRECOMPILED_DLL}")
